//= Ecos Tejidos
//===== Description: =========================================
//= La Tejedora de Ecos invoca ecos de MVPs pagando Almas y sus items.
//= Al morir el eco, todos despiertan junto a la Tejedora y el sueño se cierra.
//============================================================

function	script	F_EchoItemName	{
	set .@id, getarg(0);
	switch (.@id) {
	case 968: return "Orc Hero's Token";
	case 6326: return "Piece of Queen's Wing";
	case 7114: return "Mask of Tutankhamen";
	case 7169: return "Taeguk Plate";
	case 7211: return "Rosetta Stone Piece";
	case 25508: return "Orc Lord's Token";
	}
	return getitemname(.@id);
}

prontera,173,152,4	script	Tejedora de Ecos	4_F_THAISHAMAN,{
	.@instance_id = instance_id(IM_PARTY);

	mes "[ ^0065DFTejedora de Ecos^000000 ]";
	if (.@instance_id > 0) {
		mes "El telar ya está tenso. Un sueño sigue abierto.";
		mes "Puedes entrar mientras el sueño siga abierto.";
		next;
		.@sel = prompt("Entrar:Salir");
		if (.@sel == 255 || .@sel == 2) close;

		.@result = instance_enter("Ecos Tejidos", -1, -1, getcharid(0), .@instance_id);
		if (.@result != IE_OK) {
			mes "[ ^0065DFTejedora de Ecos^000000 ]";
			mes "El sueño no te aceptó (error " + .@result + ").";
			mes "Avisa al GM para que lo revise.";
		}
		close;
	}

	mes "No puedo invocar a la bestia, pero puedo llamar a su eco.";
	mes "Las Almas sostienen el sueño; los objetos le dan forma.";
	mes "Elige la fuerza del telar:";
	next;
	.@mode = prompt("3 Almas:5 Almas:10 Almas:Salir");
	if (.@mode == 255 || .@mode == 4) close;

	if (.@mode == 1) {
		.@start = 0;
		.@count = 10;
		.@cost = 3;
	} else if (.@mode == 2) {
		.@start = 10;
		.@count = 16;
		.@cost = 5;
	} else {
		.@start = 26;
		.@count = 3;
		.@cost = 10;
	}

	.@menu$ = "";
	for (.@i = .@start; .@i < .@start + .@count; .@i++) {
		if (.@menu$ != "") .@menu$ = .@menu$ + ":";
		.@menu$ = .@menu$ + .mvp_name$[.@i];
	}
	.@opt = prompt(.@menu$);
	if (.@opt == 255) close;
	.@idx = .@start + .@opt - 1;

	mes "[ ^0065DFTejedora de Ecos^000000 ]";
	mes "Para llamar al eco de ^0065DF" + .mvp_name$[.@idx] + "^000000 necesitas:";
	mes .@cost + " Almas de MVP (tienes " + countitem(35005) + ")";
	next;

	mes "[ ^0065DFTejedora de Ecos^000000 ]";
	mes .key1_qty[.@idx] + " " + F_EchoItemName(.key1_id[.@idx]) + " (tienes " + countitem(.key1_id[.@idx]) + ")";
	if (.key2_id[.@idx] > 0)
		mes .key2_qty[.@idx] + " " + F_EchoItemName(.key2_id[.@idx]) + " (tienes " + countitem(.key2_id[.@idx]) + ")";
	if (.key3_id[.@idx] > 0)
		mes .key3_qty[.@idx] + " " + F_EchoItemName(.key3_id[.@idx]) + " (tienes " + countitem(.key3_id[.@idx]) + ")";
	next;

	.@ready = 1;
	if (countitem(35005) < .@cost) .@ready = 0;
	if (countitem(.key1_id[.@idx]) < .key1_qty[.@idx]) .@ready = 0;
	if (.key2_id[.@idx] > 0) {
		if (countitem(.key2_id[.@idx]) < .key2_qty[.@idx]) .@ready = 0;
	}
	if (.key3_id[.@idx] > 0) {
		if (countitem(.key3_id[.@idx]) < .key3_qty[.@idx]) .@ready = 0;
	}
	if (.@ready == 0) {
		mes "[ ^0065DFTejedora de Ecos^000000 ]";
		mes "Todavía te falta:";
		next;

		mes "[ ^0065DFTejedora de Ecos^000000 ]";
		if (countitem(35005) < .@cost) mes "- " + (.@cost - countitem(35005)) + " Almas de MVP";
		if (countitem(.key1_id[.@idx]) < .key1_qty[.@idx]) mes "- " + (.key1_qty[.@idx] - countitem(.key1_id[.@idx])) + " " + F_EchoItemName(.key1_id[.@idx]);
		if (.key2_id[.@idx] > 0) {
			if (countitem(.key2_id[.@idx]) < .key2_qty[.@idx]) mes "- " + (.key2_qty[.@idx] - countitem(.key2_id[.@idx])) + " " + F_EchoItemName(.key2_id[.@idx]);
		}
		if (.key3_id[.@idx] > 0) {
			if (countitem(.key3_id[.@idx]) < .key3_qty[.@idx]) mes "- " + (.key3_qty[.@idx] - countitem(.key3_id[.@idx])) + " " + F_EchoItemName(.key3_id[.@idx]);
		}
		close;
	}

	.@sel = prompt("Tejer el sueño:Salir");
	if (.@sel == 255 || .@sel == 2) close;

	.@instance_id = instance_create("Ecos Tejidos", IM_PARTY);
	if (.@instance_id < 0) {
		mes "[ ^0065DFTejedora de Ecos^000000 ]";
		mes "El telar no respondió (error " + .@instance_id + ").";
		mes "Avisa al GM para que lo revise.";
		close;
	}

	delitem 35005, .@cost;
	delitem .key1_id[.@idx], .key1_qty[.@idx];
	if (.key2_id[.@idx] > 0) delitem .key2_id[.@idx], .key2_qty[.@idx];
	if (.key3_id[.@idx] > 0) delitem .key3_id[.@idx], .key3_qty[.@idx];

	setinstancevar 'mvp_id, .mvp_id[.@idx], .@instance_id;
	donpcevent instance_npcname("OzroTestSpawner", .@instance_id) + "::OnSummon";

	.@result = instance_enter("Ecos Tejidos", -1, -1, getcharid(0), .@instance_id);
	if (.@result != IE_OK) {
		mes "[ ^0065DFTejedora de Ecos^000000 ]";
		mes "El sueño se abrió, pero no pudiste entrar (error " + .@result + ").";
		mes "Avisa al GM para que lo revise.";
		close;
	}
	close;

OnInit:
	setarray .mvp_id[0], 1086, 1159, 1150, 1389, 1688, 1087, 1059, 1147, 1046, 1190, 1251, 1039, 1630, 1112, 1157, 1115, 1623, 1511, 1418, 1038, 1272, 1583, 1252, 2068, 1312, 1785, 2087, 1734, 2202;

	setarray .key1_id[0], 969, 1015, 1022, 732, 5116, 25508, 526, 1096, 724, 968, 603, 923, 25622, 2287, 7113, 1029, 7327, 7211, 7169, 930, 7005, 7300, 7036, 7444, 7070, 7563, 6326, 7319, 962;
	setarray .key1_qty[0], 1, 10, 20, 1, 1, 2, 30, 30, 5, 1, 5, 2, 1, 1, 2, 5, 2, 3, 2, 10, 10, 1, 2, 1, 10, 5, 3, 100, 10;

	end;
}

quiz_00,51,35,0	script	OzroTestSpawner	-1,{
OnInstanceInit:
	.@map$ = instance_mapname("quiz_00");
	removemapflag .@map$, mf_noskill;
	setmapflag .@map$, mf_nobranch;
	setmapflag .@map$, mf_nomemo;
	setmapflag .@map$, mf_noteleport;
	setmapflag .@map$, mf_monster_noteleport;
	setmapflagnosave .@map$, "prontera", 171, 149;
	end;

OnSummon:
	.@instance_id = instance_id();
	.@mob_id = getinstancevar('mvp_id, .@instance_id);
	if (.@mob_id <= 0) end;
	.@name$ = getmonsterinfo(.@mob_id, MOB_NAME);
	monster instance_mapname("quiz_00"), 81, 8, .@name$, .@mob_id, 1, instance_npcname("OzroTestSpawner", .@instance_id) + "::OnMvpDead";
	end;

OnMvpDead:
	.@instance_id = instance_id();
	instance_warpall "prontera", 171, 149, .@instance_id;
	initnpctimer;
	end;

OnTimer1000:
	stopnpctimer;
	instance_destroy instance_id();
	end;
}
