//===== OzRo Script ==========================================
//= Ludovico el Sepulturero + contador global de muertes
//===== Description: =========================================
//= Narra las muertes del personaje y estadísticas del reino.
//============================================================

-	script	ozro_death_tracker	-1,{
OnPCDieEvent:
	set $OZRO_DEATHS_TOTAL, $OZRO_DEATHS_TOTAL + 1;
	end;
}

prontera,274,352,3	script	Ludovico	1_M_PASTOR,{

	.@deaths = PC_DIE_COUNTER;
	.@global = $OZRO_DEATHS_TOTAL;

	mes "^FF0000[Ludovico el Sepulturero]^000000";
	mes "Cada vez que un alma se separa de su vasija, deja un eco frío en este mundo.";
	mes "Yo soy quien escucha esos ecos.";
	next;

	switch (select("Mis muertes", "Ecos del reino", "Irme")) {
	case 1:
		mes "^FF0000[Ludovico]^000000";
		if (.@deaths == 0) {
			mes "Tu hilo de vida es notablemente fuerte, viajero.";
			mes "Ni un solo nudo, ni una sola ruptura.";
			mes "Que los dioses lo mantengan así.";
		} else if (.@deaths <= 10) {
			mes "Tu destino está marcado por la lucha y la perseverancia.";
			mes "Veo el peso de ^FF0000" + .@deaths + " muertes^000000 sobre tus hombros.";
			mes "No te avergüences. Cada retorno del abismo es una victoria.";
		} else if (.@deaths <= 50) {
			mes "Has caído muchas veces, y aún así sigues en pie.";
			mes "Llevas ^FF0000" + .@deaths + " muertes^000000 en tu registro.";
			mes "Los veteranos saben que el abismo también enseña.";
		} else {
			mes "Por los dioses... ^FF0000" + .@deaths + " muertes^000000.";
			mes "Tu alma conoce el camino de vuelta mejor que nadie.";
			mes "Hela misma debe reconocerte al ver llegar.";
		}
		break;
	case 2:
		mes "^FF0000[Ludovico]^000000";
		mes "En todo el reino he contado ^FF0000" + .@global + " ecos de muerte^000000.";
		mes "Cada caída de un aventurero deja huella en la tierra.";
		if (.@global == 0)
			mes "Aún no he sentido ninguna... un tiempo de paz, quizás.";
		break;
	case 3:
		mes "^FF0000[Ludovico]^000000";
		mes "Que tu hilo de vida se mantenga firme, viajero.";
		break;
	}
	close;
}
