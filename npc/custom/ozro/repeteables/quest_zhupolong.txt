lou_dun02,161,264,4	script	Dragon Hunter	732,{

	if (checkquest(60182) != -1) {
		if (checkquest(60182,HUNTING) != 2 ) {
			mes "[Dragon Hunter]";
			mes "¿Ya terminaste de cazar los 50 Zhu Po Long?";
			next;
			switch(select("Sí:No:Quiero renunciar")) {
			case 1:
				mes "[Dragon Hunter]";
				mes "Mmm, no creo que hayas cazado 50 todavía.";
				close;
			case 2:
				mes "[Dragon Hunter]";
				mes "Recuerda, necesito ayuda para cazar Zhu Po Long, así que ve y caza 50 de ellos.";
				close;
			case 3:
				mes "[Dragon Hunter]";
				mes "¿Estás seguro de que quieres dejar de cazar?";
				mes "Cualquier progreso que hayas hecho se borrará";
				next;
				switch(select("Sí:No")) {
				case 1:
					mes "[Dragon Hunter]";
					mes "Está bien, vuelve aquí si cambias de opinión.";
					erasequest 60182;
					close;
				case 2:
					mes "[Dragon Hunter]";
					mes "Por favor, mata 50 Zhu Po Long para hacer de este un lugar al que la gente pueda ir a ganar Experiencia.";
					close;
				}
			}
		}
		mes "[Dragon Hunter]";
		mes "Increíble, lo hiciste con rapidez.";
		getexp 151300,60520;
		erasequest 60182;
		close;
	}
	if ((BaseLevel > 49) && (BaseLevel < 91)) {
		mes "[Dragon Hunter]";
		mes "¡Genial!";
		mes "Estás aquí para ayudarme, ¿verdad?";
		mes "Si no, ¡no sé qué hacer con los monstruos en este lugar!";
		next;
		switch(select("Claro:No")) {
		case 1:
			mes "[Dragon Hunter]";
			mes "¡Genial! Supe al verte que puedo contar contigo.";
			mes "¡Necesito que reduzcas los monstruos que arrasan este mapa!";
			mes "Hay demasiados Zhu Po Long aquí: los niveles bajos no ganan buena experiencia.";
			next;
			mes "[Dragon Hunter]";
			mes "¿Todavía quieres ayudarme?";
			next;
			switch(select("Claro:Cancelar")) {
			case 1:
				if (BaseLevel > 90) {
					mes "[Dragon Hunter]";
					mes "¡Gracias por tu gran ayuda con los Zhu Po Long!";
					mes "Actualmente la situación es estable y ya no necesito tu ayuda para matarlos.";
					close;
				}
				mes "[Dragon Hunter]";
				mes "Mata 50 de esos Zhu Po Long y avísame cuando termines. Te recompensaré por tus esfuerzos.";
				setquest 60182;
				close;
			case 2:
				mes "[Dragon Hunter]";
				mes "Si cambias de opinión, vuelve, por favor.";
				close;
			}
		case 2:
			mes "[Dragon Hunter]";
			mes "Si cambias de opinión, vuelve, por favor.";
			close;
		}
	}
	mes "[Dragon Hunter]";
	mes "¡Ay, ay!";
	next;
	mes "[Dragon Hunter]";
	mes "Lo siento, no puedo hablar ahora. Esos Zhu Po Long van a matarme.";
	close;
}
