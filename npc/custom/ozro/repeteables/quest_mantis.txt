lou_fild01,240,354,4	script	Mantis Researcher	878,{

	if (checkquest(60179) != -1) {
		if (checkquest(60179,HUNTING) != 2 ) {
			mes "[Mantis Researcher]";
			mes "¿Ya terminaste de cazar los 50 Mantis?";
			next;
			switch(select("Sí:No:Quiero renunciar")) {
			case 1:
				mes "[Mantis Researcher]";
				mes "Mmm, no creo que hayas cazado 50 todavía.";
				close;
			case 2:
				mes "[Mantis Researcher]";
				mes "Recuerda, necesito ayuda para cazar Mantis, así que ve y caza 50 de ellos.";
				close;
			case 3:
				mes "[Mantis Researcher]";
				mes "¿Estás seguro de que quieres dejar de cazar?";
				mes "Cualquier progreso que hayas hecho se borrará";
				next;
				switch(select("Sí:No")) {
				case 1:
					mes "[Mantis Researcher]";
					mes "Está bien, vuelve aquí si cambias de opinión.";
					erasequest 60179;
					close;
				case 2:
					mes "[Mantis Researcher]";
					mes "Por favor, mata 50 Mantis para hacer de este un lugar al que la gente pueda ir a ganar Experiencia.";
					close;
				}
			}
		}
		mes "[Mantis Researcher]";
		mes "Increíble, lo hiciste con rapidez.";
		getexp 18818,7527;
		erasequest 60179;
		close;
	}
	if ((BaseLevel > 34) && (BaseLevel < 71)) {
		mes "[Mantis Researcher]";
		mes "¡Genial!";
		mes "Estás aquí para ayudarme, ¿verdad?";
		mes "Si no, ¡no sé qué hacer con los monstruos en este lugar!";
		next;
		switch(select("Claro:No")) {
		case 1:
			mes "[Mantis Researcher]";
			mes "¡Genial! Supe al verte que puedo contar contigo.";
			mes "¡Necesito que reduzcas los monstruos que arrasan este mapa!";
			mes "Hay demasiados Mantis aquí: los niveles bajos no pueden ganar buena experiencia.";
			next;
			mes "[Mantis Researcher]";
			mes "¿Todavía quieres ayudarme?";
			next;
			switch(select("Claro:Cancelar")) {
			case 1:
				if (BaseLevel > 70) {
					mes "[Mantis Researcher]";
					mes "¡Gracias por tu gran ayuda con los Mantis!";
					mes "Actualmente la situación es estable y ya no necesito tu ayuda para matarlos.";
					close;
				}
				mes "[Mantis Researcher]";
				mes "Mata 50 de esos Mantis y avísame cuando termines. Te recompensaré por tus esfuerzos.";
				setquest 60179;
				close;
			case 2:
				mes "[Mantis Researcher]";
				mes "Si cambias de opinión, vuelve, por favor.";
				close;
			}
		case 2:
			mes "[Mantis Researcher]";
			mes "Si cambias de opinión, vuelve, por favor.";
			close;
		}
	}
	mes "[Mantis Researcher]";
	mes "¡Ay, ay!";
	next;
	mes "[Mantis Researcher]";
	mes "Lo siento, no puedo hablar ahora. Esos Mantis van a matarme.";
	close;
}
