//===== How to Use: ==========================================
//= Uses function "F_Bounty_Board" to manage the Bounty Boards
//= npc on cities.
//= The function arguments are this:
//= 	callfunc "F_Bounty_Board", MinLevel, MaxLevel, MsgHuntPlace, MsgOutLevel,
//=			MonsterID, 		QuestID, 	BaseExp, 	JobExp{, 
//=			MonsterID, 		QuestID, 	BaseExp, 	JobExp};
//============================================================

function	script	F_Bounty_Board	{
	if (BaseLevel < getarg(0) || (BaseLevel > getarg(1) && getarg(1))){
		for (.@i = 5; .@i < getargcount(); .@i += 4){
			if (checkquest(getarg(.@i)) != -1){
				mes "[Bounty Board]";
				mes "Estás listo para cobrar tu recompensa.";
				mes "¿Qué recompensa quieres cobrar?";
				next;
				switch(select("Cobrar 1 recompensa", "Cobrar TODAS las recompensas", "Quiero DEJAR de cazar", "cancelar")){
					case 1: goto L_BountyOne;
					case 2: goto L_BountyAll;
					case 3: goto L_BountyDel;
					case 4: close;
				}
			}
		}
		switch(getarg(3)){
			case 1: mes "Estas recompensas solo están disponibles para aventureros que estén entre los niveles "+getarg(0)+"-"+getarg(1)+""; break;
			case 2: mes "Estas recompensas solo están disponibles para aventureros que estén entre los niveles "+getarg(0)+" y "+getarg(1)+""; break;
			case 3: mes "Estas recompensas solo están disponibles para aventureros que estén por encima del nivel "+getarg(0)+""; break;
			case 4: mes "Estas recompensas solo están disponibles para aventureros con un nivel superior a "+getarg(0)+""; break;
			case 5: mes "Has encontrado un tablón de anuncios lleno de carteles de búsqueda. Nada parece tan interesante."; break;
		}
		close;
	}
	mes "[Bounty Board]";
	mes "Has encontrado un tablón de anuncios lleno de carteles de búsqueda.";
	mes "¿Quieres mirar más de cerca?";
	next;
	mes "[Bounty Board]";
	mes "Hay muchos anuncios de caza de monstruos.";
	next;
	switch(select("Revisar misiones de recompensa", "Cobrar recompensas", "¿Dónde voy a cazar?", "Busca a otros cazadores de recompensas.", "Quiero DEJAR de cazar", "No me interesa.")){
	case 1:	// Check bounty missions
		mes "[Bounty Board]";
		mes "Hay una lista de monstruos en el tablón.";
		mes "Al revisarla, decides aceptar una tarea de caza.";
		next;
		for (.@i = 4; .@i < getargcount(); .@i += 4)
			.@menu$ = .@menu$ + getmonsterinfo(getarg(.@i),MOB_NAME) + ":";
		.@menu$ = .@menu$ + "Cazar todo:Cancelar";
		.@bounty = (select(.@menu$) * 4) + 1;
		if (.@bounty > getargcount()+1){
			mes "[Bounty Board]";
			mes "Has decidido no aceptar ninguna recompensa.";
			close;
		} else if (.@bounty == getargcount()+1){
			for (.@i = 5; .@i < getargcount(); .@i += 4){
				if (checkquest(getarg(.@i)) == -1)
					setquest getarg(.@i);
				.@HuntAllList$ = .@HuntAllList$ + (.@i+4 > getargcount()?" and ":(.@i > 5?", ":"")) + getmonsterinfo(getarg(.@i-1),MOB_NAME);
			}
			mes "[Bounty Board]";
			mes "Hay una recompensa por cazar 150 " + .@HuntAllList$ + " cazarlos a todos.";
			close;
		}
		.@monster$ = getmonsterinfo(getarg(.@bounty-1),MOB_NAME);
		mes "[Bounty Board]";
		mes "Hay una recompensa por cazar 150 " + .@monster$ + "";
		next;
		if (select("Lo haré.", "Uhh, en realidad déjalo.") == 1){
			if (checkquest(getarg(.@bounty),HUNTING) == -1)
				setquest getarg(.@bounty);
			mes "[Bounty Board]";
			mes "Has aceptado esta recompensa:";
			mes .@monster$ + "";
		}
		close;
	case 2:	// Collect bounty rewards
		mes "[Bounty Board]";
		mes "Estás listo para cobrar tu recompensa.";
		mes "¿Qué recompensa quieres cobrar?";
		next;
		switch(select("Cobrar 1 recompensa", "Cobrar TODAS las recompensas", "cancelar")){
			case 1: goto L_BountyOne;
			case 2: goto L_BountyAll;
			case 3: close;
		}
	case 3:	// Where do I go to hunt?
		mes "[Bounty Board]";
		switch(getarg(2)){
			case 1: mes "Puedes cazarlos en la mazmorra. ¡Está justo frente a ti!"; break;
			case 2: mes "Puedes cazarlos en los bosques que rodean Payon y en la Cueva al norte de Archer Village."; break;
			case 3: mes "Puedes cazarlos en varios campos que rodean Prontera."; break;
			case 4: mes "Puedes cazarlos en varios campos y mazmorras alrededor de Prontera."; break;
			case 5: mes "Puedes cazar en los Campos de Umbala para cumplir estas recompensas."; break;
			case 6:
				mes "¿A qué campo de caza de monstruos quieres ir?";
				next;
				.@bounty = (select("Golem", "Stalactic Golem", "Medusa", "Cancelar") * 4) + 1;
				if (.@bounty > getargcount())
					close;
				mes "[Bounty Board]";
				if (checkquest(getarg(.@bounty),HUNTING) == -1){
					mes "Actualmente no estás cazando "+getmonsterinfo(getarg(.@bounty-1),MOB_NAME)+"";
					close;
				}
				mes "Enviándote ahora.";
				close2;
				switch(.@bounty){
					case 1: warp "cmd_fild06",147,36; break;
					case 2: warp "beach_dun2",154,25; break;
					case 3: warp "beach_dun",266,67; break;
				}
				end;
			case 7: mes "Puedes cazar en <NAVI>Juno Fields<INFO>yuno,158,16,</INFO></NAVI> para cumplir estas recompensas."; break;
			case 8: mes "Puedes cazar en <NAVI>Brasilis Field<INFO>brasilis,309,304,</INFO></NAVI> y en <NAVI>Brasilis Dungeon<INFO>brasilis,183,284,</INFO></NAVI> para cumplir estas recompensas."; break;
			case 9: mes "Puedes cazarlos en varios campos que rodean Lighthalzen."; break;
			case 10: mes "Puedes cazarlos en varios campos y mazmorras alrededor de Lighthalzen."; break;
			case 11:
				mes "Puedes cazar en <NAVI>Splendide Field<INFO>mid_camp,14,143,</INFO></NAVI> para cumplir estas recompensas.";
				mes "Asegúrate de ser un miembro autorizado de la Alianza de Midgard antes de ir allí.";
				break;
			case 12: mes "Puedes cazar en los dos Campos de Dicastes para cumplir estas recompensas."; break;
		}
		close;
	case 4:	// Search for other bounty hunters.
		mes "[Bounty Board]";
		mes "Para encontrar jugadores con quienes formar grupo, haz clic en el botón [Booking] de tu menú de usuario o escribe \"/booking\" en el cuadro de chat para abrir la ventana.";
		next;
		mes "[Bounty Board]";
		mes "Luego haz clic en \"Search\" para ver todos los anuncios de grupos.";
		mes "Para publicar un anuncio de búsqueda de grupo, escribe \"/recruit\" en el cuadro de chat.";
		close;
	case 5:	// I want to QUIT hunting
		goto L_BountyDel;
	case 6:	// Not interested.
		mes "[Bounty Board]";
		mes "La vida está llena de peligros.";
		mes "No vives la vida a menos que te pongas en peligro de vez en cuando.";
		close;
	}

L_BountyOne:
	mes "[Bounty Board]";
	mes "Recibirás ^FF00001 Eden Merit Badge si cobras todas juntas.^000000";
	mes "Si no, zeny y EXP/JEXP se entregan por separado.";
	mes "^FF0000Las cuentas VIP eligen cómo repartir la experiencia^000000.";
	next;
	for (.@i = 4; .@i < getargcount(); .@i += 4)
		.@menu$ = .@menu$ + getmonsterinfo(getarg(.@i),MOB_NAME) + ":";
	.@menu$ = .@menu$ + "Cancelar";
	.@bounty = (select(.@menu$) * 4) + 1;
	if (.@bounty > getargcount()){
		mes "^000000[Bounty Board]^000000";
		mes "Vuelve cuando tengas una recompensa completada.";
		close;
	}
	if (checkquest(getarg(.@bounty),HUNTING) == 2){
		if (vip_status(1)){
			mes "^000000[Bounty Board]^000000";
			mes "¿Cómo distribuirás tu nueva experiencia?";
			next;
			.@vip_reward = select("^FF0000Tanto Base EXP como Job EXP^000000", "^0000FFSolo quiero Base EXP.^000000", "^FFA400Solo quiero Job EXP.^000000");
		}
		erasequest getarg(.@bounty);
		if (Zeny + 37500 <= MAX_ZENY)
			Zeny += 37500;
		.@BaseExp = getarg(.@bounty+1);
		.@JobExp = getarg(.@bounty+2);
		switch(.@vip_reward){
			case 2:
				getexp .@BaseExp,0;
				getexp .@JobExp,0;
				break;
			case 3:
				getexp 0,.@BaseExp;
				getexp 0,.@JobExp;
				break;
			default:
				getexp .@BaseExp,.@JobExp;
				break;
		}
	} else {
		mes "^000000[Bounty Board]^000000";
		mes "La recompensa por cazar "+getmonsterinfo(getarg(.@bounty-1),MOB_NAME)+" no se ha completado.";
	}
	close;

L_BountyAll:
	mes "[Bounty Board]";
	mes "Vaya, ¿te crees un buen cazador, eh?";
	mes "Déjame ver si realmente has cazado todos los monstruos que dijiste.";
	next;
	for (.@i = 5; .@i < getargcount(); .@i += 4){
		if (checkquest(getarg(.@i),HUNTING) < 2){
			mes "[Bounty Board]";
			mes "La recompensa por cazar todo no se ha completado.";
			mes "No se tolerarán las mentiras.";
			close;
		}
	}
	if (vip_status(1)){
		mes "^000000[Bounty Board]^000000";
		mes "¿Cómo distribuirás tu nueva experiencia?";
		next;
		.@vip_reward = select("^FF0000Tanto Base EXP como Job EXP^000000", "^0000FFSolo quiero Base EXP.^000000", "^FFA400Solo quiero Job EXP.^000000");
	}
	for (.@i = 5; .@i < getargcount(); .@i += 4)
		erasequest getarg(.@i);
	for (.@i = 6; .@i < getargcount(); .@i += 4){
		if (Zeny + 37500 <= MAX_ZENY)
			Zeny += 37500;
		.@BaseExp = getarg(.@i);
		.@JobExp = getarg(.@i+1);
		switch(.@vip_reward){
			case 2:
				getexp .@BaseExp,0;
				getexp .@JobExp,0;
				break;
			case 3:
				getexp 0,.@BaseExp;
				getexp 0,.@JobExp;
				break;
			default:
				getexp .@BaseExp,.@JobExp;
				break;
		}
	}
	getitem "Para_Logro_Badge",1;
	mes "[Bounty Board]";
	mes "Aquí está tu recompensa.";
	close;

L_BountyDel:
	mes "[Bounty Board]";
	mes "Revisaré qué misiones has aceptado hasta ahora.";
	mes "¿Cuáles quieres abandonar?";
	next;
	for (.@i = 5; .@i < getargcount(); .@i += 4){
		if (checkquest(getarg(.@i),HUNTING) == 1){
			.@monster$ = getmonsterinfo(getarg(.@i-1),MOB_NAME);
			mes "[Bounty Board]";
			mes "NO has completado la " + .@monster$ + " misión de caza.";
			mes "¿Quieres continuar o abandonar?";
			next;
			if (select("Sigue cazando "+.@monster$, "^FF0000Deshazte de mi misión^000000") == 2){
				erasequest getarg(.@i);
				mes "[Bounty Board]";
				mes "Todos tus registros de esta misión se han borrado.";
				close2;
				.@questerased = 1;
			}
		}
	}
	if (!.@questerased)
		close;
	else
		end;

}
