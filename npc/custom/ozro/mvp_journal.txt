//===== OzRo Script ==========================================
//= El Guardián de Almas
//===== Description: =========================================
//= Lee MVP_KILL_<id> y MVP_SUPP_<id> y cristaliza fragmentos en Almas.
//= Muestra resumen, lista por gran bestia y permite reclamar Almas.
//= Los puntos de Alma se calculan desde las variables viejas.
//============================================================

prontera,150,180,4	script	El Guardián de Almas	4_M_SURA,{

	mes .npc_name$;
	mes "No todas las bestias se van cuando mueren.";
	mes "Sus almas se niegan a abandonar el mundo, y sus fragmentos se adhieren a quienes las mataron.";
	next;

	mes .npc_name$;
	mes "Puedo ver esos fragmentos en ti. ¿Qué deseas hacer?";
	mes "Si quieres volver a enfrentar a una bestia, busca a la Tejedora. Ella sabe llamar a su eco.";
	next;

	switch (select("Resumen:Mis bestias:Almas:Salir")) {
	case 1:
		callsub L_Summary;
		break;
	case 2:
		callsub L_List;
		break;
	case 3:
		callsub L_Souls;
		break;
	case 4:
		mes .npc_name$;
		mes "Que las almas no te arrastren.";
		break;
	}
	close;

L_Summary:
	callsub L_CalcStats;
	callsub L_CalcSouls;
	mes .npc_name$;
	mes "^777777Tus fragmentos de alma^000000";
	mes "Golpes finales: ^009900" + @MVPJ_KILLS + "^000000";
	mes "Asistencias: ^009900" + @MVPJ_SUPPS + "^000000";
	mes "Grandes bestias únicas: ^009900" + @MVPJ_UNIQUE + " / " + .mvp_count + "^000000";
	mes "Fragmentos de alma pendientes: ^009900" + @MVPJ_PENDING + "^000000";
	next;

	mes .npc_name$;
	if (@MVPJ_TOP_KILL_ID > 0) {
		mes "Bestia más cazada: ^FF0000" + getmonsterinfo(@MVPJ_TOP_KILL_ID, MOB_NAME) + "^000000";
		mes "(" + @MVPJ_TOP_KILL_COUNT + " golpes finales)";
	}
	if (@MVPJ_TOP_SUPP_ID > 0) {
		mes "Bestia más asistida: ^0000FF" + getmonsterinfo(@MVPJ_TOP_SUPP_ID, MOB_NAME) + "^000000";
		mes "(" + @MVPJ_TOP_SUPP_COUNT + " asistencias)";
	}
	if (@MVPJ_KILLS == 0 && @MVPJ_SUPPS == 0) {
		mes "Aún no has derrotado ninguna gran bestia.";
		mes "Vuelve cuando enfrentes a una.";
	}
	next;
	return;

L_List:
	callsub L_CalcStats;
	if (@MVPJ_UNIQUE == 0) {
		mes .npc_name$;
		mes "Todavía no veo fragmentos en ti.";
		mes "Vuelve después de enfrentar a una gran bestia.";
		next;
		return;
	}

	.@pages = (@MVPJ_UNIQUE + 9) / 10;
	.@page = 0;
	while (1) {
		mes .npc_name$;
		mes "^777777Tus grandes bestias - página " + (.@page + 1) + "/" + .@pages + "^000000";
		.@shown = 0;
		.@seen = 0;
		for (.@i = 0; .@i < .mvp_count; .@i++) {
			.@id = .mvp_id[.@i];
			.@k = getd("MVP_KILL_" + .@id);
			.@s = getd("MVP_SUPP_" + .@id);
			if (.@k == 0 && .@s == 0) continue;
			if (.@seen < .@page * 10) {
				.@seen = .@seen + 1;
				continue;
			}
			if (.@shown >= 10) break;
			mes getmonsterinfo(.@id, MOB_NAME) + " - Golpes: " + .@k + " / Asistencias: " + .@s;
			.@shown = .@shown + 1;
			.@seen = .@seen + 1;
		}
		next;
		.@sel = select("Siguiente:Anterior:Volver");
		if (.@sel == 1) {
			if (.@page + 1 < .@pages) {
				.@page = .@page + 1;
			} else {
				mes .npc_name$;
				mes "Ya estás en la última página.";
				next;
			}
		} else if (.@sel == 2) {
			if (.@page > 0) {
				.@page = .@page - 1;
			} else {
				mes .npc_name$;
				mes "Ya estás en la primera página.";
				next;
			}
		} else {
			return;
		}
	}

L_Souls:
	callsub L_CalcSouls;
	mes .npc_name$;
	mes "^777777Almas de las grandes bestias^000000";
	mes "Fragmentos ganados: ^009900" + @MVPJ_EARNED + "^000000";
	mes "Almas reclamadas: ^009900" + MVP_SOUL_CLAIMED + "^000000";
	next;

	mes .npc_name$;
	mes "Fragmentos pendientes: ^009900" + @MVPJ_PENDING + "^000000";
	mes "Cada fragmento se convierte en 1 Alma de MVP.";
	next;
	if (@MVPJ_PENDING <= 0) {
		mes .npc_name$;
		mes "No tienes fragmentos pendientes.";
		mes "Sigue enfrentando grandes bestias.";
		next;
		return;
	}
	if (select("Reclamar Almas:Volver") == 2) return;
	if (!checkweight(35005, @MVPJ_PENDING)) {
		mes .npc_name$;
		mes "No tienes espacio suficiente para " + @MVPJ_PENDING + " Almas.";
		mes "Haz lugar en el inventario y vuelve a intentarlo.";
		next;
		return;
	}
	getitem 35005, @MVPJ_PENDING;
	MVP_SOUL_CLAIMED = MVP_SOUL_CLAIMED + @MVPJ_PENDING;
	mes .npc_name$;
	mes "Reclamaste " + @MVPJ_PENDING + " Almas de MVP.";
	next;
	return;

L_CalcStats:
	@MVPJ_KILLS = 0;
	@MVPJ_SUPPS = 0;
	@MVPJ_UNIQUE = 0;
	@MVPJ_TOP_KILL_ID = 0;
	@MVPJ_TOP_KILL_COUNT = 0;
	@MVPJ_TOP_SUPP_ID = 0;
	@MVPJ_TOP_SUPP_COUNT = 0;

	for (.@i = 0; .@i < .mvp_count; .@i++) {
		.@id = .mvp_id[.@i];
		.@k = getd("MVP_KILL_" + .@id);
		.@s = getd("MVP_SUPP_" + .@id);
		@MVPJ_KILLS = @MVPJ_KILLS + .@k;
		@MVPJ_SUPPS = @MVPJ_SUPPS + .@s;
		if (.@k > 0 || .@s > 0) @MVPJ_UNIQUE = @MVPJ_UNIQUE + 1;
		if (.@k > @MVPJ_TOP_KILL_COUNT) {
			@MVPJ_TOP_KILL_ID = .@id;
			@MVPJ_TOP_KILL_COUNT = .@k;
		}
		if (.@s > @MVPJ_TOP_SUPP_COUNT) {
			@MVPJ_TOP_SUPP_ID = .@id;
			@MVPJ_TOP_SUPP_COUNT = .@s;
		}
	}
	return;

L_CalcSouls:
	@MVPJ_EARNED = 0;
	for (.@i = 0; .@i < .mvp_count; .@i++) {
		.@id = .mvp_id[.@i];
		.@k = getd("MVP_KILL_" + .@id);
		.@s = getd("MVP_SUPP_" + .@id);
		if (.@k + .@s <= 0) continue;

		switch (.@id) {
		case 3505: case 1086: case 1159: case 1150: case 1087: case 1147: case 1389: case 1688: case 1059: case 1046: case 1190:
			.@points = 1;
			break;
		case 1251: case 1039: case 1630: case 1112: case 1157: case 1492: case 1115: case 1623: case 1511: case 1418: case 1885: case 1038: case 1272: case 1583: case 2156: case 1252: case 2068: case 1312: case 1785:
			.@points = 3;
			break;
		case 2441: case 2442: case 2249: case 2087: case 2253: case 1734: case 2255: case 2251: case 2362: case 3633: case 1658: case 1768: case 1685: case 2483: case 1917: case 2202:
			.@points = 5;
			break;
		case 20381: case 3741: case 20648: case 3074: case 1832: case 1751: case 2165: case 1719: case 1871:
			.@points = 10;
			break;
		case 20601: case 20419: case 20422: case 20421: case 21301: case 20934: case 20928: case 20943:
			.@points = 20;
			break;
		default:
			.@points = 0;
			break;
		}
		@MVPJ_EARNED = @MVPJ_EARNED + ((.@k + .@s) * .@points);
	}
	@MVPJ_PENDING = @MVPJ_EARNED - MVP_SOUL_CLAIMED;
	if (@MVPJ_PENDING < 0) @MVPJ_PENDING = 0;
	return;

OnInit:
	.npc_name$ = "[ ^0000FFEl Guardián de Almas^000000 ]";
	setarray .mvp_id[0], 1511, 2362, 2249, 1785, 1039, 3505, 2068, 20648, 21301, 20422, 20421, 2255, 1272, 20943, 1719, 1046, 1389, 1112, 1115, 1658, 1418, 1871, 2253, 2251, 1768, 2165, 1086, 1885, 1252, 1832, 20601, 1734, 2442, 2202, 1688, 2156, 1147, 1059, 1150, 2483, 1087, 1190, 1038, 1157, 1159, 20934, 20381, 20419, 1623, 1492, 2087, 3741, 1251, 1583, 2441, 20928, 3074, 1312, 1751, 3633, 1685, 1630, 1917;
	.mvp_count = getarraysize(.mvp_id);
	end;
}
