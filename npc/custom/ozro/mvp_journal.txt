//===== OzRo Script ==========================================
//= Cronista de Cacerias
//===== Description: =========================================
//= Lee MVP_KILL_<id> y MVP_SUPP_<id>.
//= Muestra resumen, lista por MVP y permite reclamar Almas.
//= Los puntos de Alma se calculan desde las variables viejas.
//============================================================

prontera,150,180,4	script	Cronista de Cacerias	4_M_SAGE_C,{

	mes .npc_name$;
	mes "Saludos, cazador. Puedo leer las historias escritas en tus cicatrices.";
	mes "¿Qué deseas saber?";
	next;

	switch (select("Resumen:Mis MVPs:Almas:Salir")) {
	case 1:
		callsub L_Summary;
		break;
	case 2:
		callsub L_List;
		break;
	case 3:
		callsub L_Souls;
		break;
	case 4:
		mes .npc_name$;
		mes "Que la caza te sea favorable.";
		break;
	}
	close;

L_Summary:
	callsub L_CalcStats;
	callsub L_CalcSouls;
	mes .npc_name$;
	mes "^777777Tus estadísticas de caza^000000";
	mes "Golpes finales: ^009900" + $@MVPJ_KILLS + "^000000";
	mes "Asistencias: ^009900" + $@MVPJ_SUPPS + "^000000";
	mes "MVPs únicos: ^009900" + $@MVPJ_UNIQUE + " / " + .mvp_count + "^000000";
	mes "Puntos de Alma pendientes: ^009900" + $@MVPJ_PENDING + "^000000";
	if ($@MVPJ_TOP_KILL_ID > 0) {
		mes "MVP más cazado: ^FF0000" + getmonsterinfo($@MVPJ_TOP_KILL_ID, MOB_NAME) + "^000000";
		mes "(" + $@MVPJ_TOP_KILL_COUNT + " golpes finales)";
	}
	if ($@MVPJ_TOP_SUPP_ID > 0) {
		mes "MVP más asistido: ^0000FF" + getmonsterinfo($@MVPJ_TOP_SUPP_ID, MOB_NAME) + "^000000";
		mes "(" + $@MVPJ_TOP_SUPP_COUNT + " asistencias)";
	}
	if ($@MVPJ_KILLS == 0 && $@MVPJ_SUPPS == 0) {
		mes "Aún no has derrotado ningún MVP.";
		mes "Empieza tu viaje de caza hoy mismo.";
	}
	next;
	return;

L_List:
	callsub L_CalcStats;
	if ($@MVPJ_UNIQUE == 0) {
		mes .npc_name$;
		mes "Todavía no tengo registros tuyos.";
		mes "Vuelve después de cazar algún MVP.";
		next;
		return;
	}

	.@pages = ($@MVPJ_UNIQUE + 9) / 10;
	.@page = 0;
	while (1) {
		mes .npc_name$;
		mes "^777777Tus MVPs - página " + (.@page + 1) + "/" + .@pages + "^000000";
		.@shown = 0;
		.@seen = 0;
		for (.@i = 0; .@i < .mvp_count; .@i++) {
			.@id = .mvp_id[.@i];
			.@k = getd("MVP_KILL_" + .@id);
			.@s = getd("MVP_SUPP_" + .@id);
			if (.@k == 0 && .@s == 0) continue;
			if (.@seen < .@page * 10) {
				.@seen = .@seen + 1;
				continue;
			}
			if (.@shown >= 10) break;
			mes getmonsterinfo(.@id, MOB_NAME) + " - Kills: " + .@k + " / Asist: " + .@s;
			.@shown = .@shown + 1;
			.@seen = .@seen + 1;
		}
		next;
		.@sel = select("Siguiente:Anterior:Volver");
		if (.@sel == 1) {
			if (.@page + 1 < .@pages) {
				.@page = .@page + 1;
			} else {
				mes .npc_name$;
				mes "Ya estás en la última página.";
				next;
			}
		} else if (.@sel == 2) {
			if (.@page > 0) {
				.@page = .@page - 1;
			} else {
				mes .npc_name$;
				mes "Ya estás en la primera página.";
				next;
			}
		} else {
			return;
		}
	}

L_Souls:
	callsub L_CalcSouls;
	mes .npc_name$;
	mes "^777777Almas de MVP^000000";
	mes "Puntos ganados: ^009900" + $@MVPJ_EARNED + "^000000";
	mes "Almas reclamadas: ^009900" + MVP_SOUL_CLAIMED + "^000000";
	mes "Puntos pendientes: ^009900" + $@MVPJ_PENDING + "^000000";
	mes "Cada punto se convierte en 1 Alma de MVP.";
	next;
	if ($@MVPJ_PENDING <= 0) {
		mes .npc_name$;
		mes "No tienes puntos pendientes.";
		mes "Sigue cazando MVPs naturales.";
		next;
		return;
	}
	if (select("Reclamar Almas:Volver") == 2) return;
	if (!checkweight(35005, $@MVPJ_PENDING)) {
		mes .npc_name$;
		mes "No tienes espacio suficiente para " + $@MVPJ_PENDING + " Almas.";
		mes "Haz lugar en el inventario y vuelve a intentarlo.";
		next;
		return;
	}
	getitem 35005, $@MVPJ_PENDING;
	MVP_SOUL_CLAIMED = MVP_SOUL_CLAIMED + $@MVPJ_PENDING;
	mes .npc_name$;
	mes "Reclamaste " + $@MVPJ_PENDING + " Almas de MVP.";
	next;
	return;

L_CalcStats:
	$@MVPJ_KILLS = 0;
	$@MVPJ_SUPPS = 0;
	$@MVPJ_UNIQUE = 0;
	$@MVPJ_TOP_KILL_ID = 0;
	$@MVPJ_TOP_KILL_COUNT = 0;
	$@MVPJ_TOP_SUPP_ID = 0;
	$@MVPJ_TOP_SUPP_COUNT = 0;

	for (.@i = 0; .@i < .mvp_count; .@i++) {
		.@id = .mvp_id[.@i];
		.@k = getd("MVP_KILL_" + .@id);
		.@s = getd("MVP_SUPP_" + .@id);
		$@MVPJ_KILLS = $@MVPJ_KILLS + .@k;
		$@MVPJ_SUPPS = $@MVPJ_SUPPS + .@s;
		if (.@k > 0 || .@s > 0) $@MVPJ_UNIQUE = $@MVPJ_UNIQUE + 1;
		if (.@k > $@MVPJ_TOP_KILL_COUNT) {
			$@MVPJ_TOP_KILL_ID = .@id;
			$@MVPJ_TOP_KILL_COUNT = .@k;
		}
		if (.@s > $@MVPJ_TOP_SUPP_COUNT) {
			$@MVPJ_TOP_SUPP_ID = .@id;
			$@MVPJ_TOP_SUPP_COUNT = .@s;
		}
	}
	return;

L_CalcSouls:
	$@MVPJ_EARNED = 0;
	for (.@i = 0; .@i < .mvp_count; .@i++) {
		.@id = .mvp_id[.@i];
		.@k = getd("MVP_KILL_" + .@id);
		.@s = getd("MVP_SUPP_" + .@id);
		if (.@k + .@s <= 0) continue;

		.@mexp = getmonsterinfo(.@id, MOB_MVPEXP);
		.@level = getmonsterinfo(.@id, MOB_LV);
		if (.@mexp > 0) {
			if (.@mexp < 500000) .@points = 1;
			else if (.@mexp < 1500000) .@points = 5;
			else .@points = 10;
		} else {
			if (.@level < 120) .@points = 1;
			else if (.@level < 180) .@points = 5;
			else .@points = 10;
		}
		$@MVPJ_EARNED = $@MVPJ_EARNED + ((.@k + .@s) * .@points);
	}
	$@MVPJ_PENDING = $@MVPJ_EARNED - MVP_SOUL_CLAIMED;
	if ($@MVPJ_PENDING < 0) $@MVPJ_PENDING = 0;
	return;

OnInit:
	.npc_name$ = "[ ^0000FFCronista de Cacerías^000000 ]";
	setarray .mvp_id[0], 1511, 2362, 2249, 1785, 1039, 3505, 2068, 20648, 21301, 20422, 20421, 2255, 1272, 20943, 1719, 1046, 1389, 1112, 1115, 1658, 1418, 1871, 2253, 2251, 1768, 2165, 1086, 1885, 1252, 1832, 20601, 1734, 2442, 2202, 1688, 2156, 1147, 1059, 1150, 2483, 1087, 1190, 1038, 1157, 1159, 20934, 20381, 20419, 1623, 1492, 2087, 3741, 1251, 1583, 2441, 20928, 3074, 1312, 1751, 3633, 1685, 1630, 1917;
	.mvp_count = getarraysize(.mvp_id);
	return;
}
