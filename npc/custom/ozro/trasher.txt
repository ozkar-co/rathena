geffen,143,70,3	script	Garrett	1_M_SIGN1,{

	.@today = gettime(DT_YYYYMMDD);
	callsub L_ResetDay, .@today;

	if (#GARRETT_MISSION_ACTIVE == 1) {
		callsub L_MissionStatus, .@today;
		if (#GARRETT_MISSION_ACTIVE == 1) close;
	}

	mes .npc_name$;
	if (OZRO_KARMA < 0) {
		mes "Tú otra vez...";
		mes "No me gusta tu aura, pero el negocio es el negocio.";
		mes "Mientras no me traigas problemas, aquí tienes un trato.";
	} else if (OZRO_KARMA > 0) {
		mes "Ah, " + strcharinfo(0) + ". Me alegra verte.";
		mes "Tienes buen ojo para lo que vale la pena, eso se nota.";
		mes "Y yo sé reconocer a quien sabe tratar la chatarra.";
	} else {
		mes "Bienvenido, forastero.";
		mes "Dicen que la basura de un hombre es el tesoro de otro.";
		mes "Yo llevo años separando lo que sirve de lo que estorba.";
	}
	next;

	mes .npc_name$;
	mes "Recorro las afueras de Geffen buscando lo que otros tiran.";
	mes "Hoy tengo " + .offer_count + " cosas en el ojo.";
	mes "Presupuesto de compras: " + F_InsertComma(.daily_cap - #GARRETT_SPENT) + " zeny.";
	next;

	switch (select("Ver ofertas del día:Misiones:Sobre tu negocio:Marcharme")) {
	case 1:
		callsub L_Offers;
		break;
	case 2:
		callsub L_MissionMenu, .@today;
		break;
	case 3:
		mes .npc_name$;
		mes "Llevo " + F_InsertComma(#GARRETT_TOTAL_SPENT) + " zeny invertidos en chatarra de aventureros.";
		mes "Contigo he cerrado " + #GARRETT_TRADES + " tratos. Cada uno cuenta.";
		mes "Misiones completadas: " + #GARRETT_MISSIONS_DONE + ".";
		break;
	case 4:
		mes .npc_name$;
		mes "Buen viaje. La basura de hoy puede ser la riqueza de mañana.";
		break;
	}
	close;

L_Offers:
	if (#GARRETT_SPENT >= .daily_cap) {
		mes .npc_name$;
		mes "Amigo, ya no me queda una moneda en la bolsa hoy.";
		mes "Vuelve mañana. Siempre hay basura nueva que vale la pena.";
		next;
		return;
	}

	.@page = 0;
	while (1) {
		.@start = .@page * .offer_page_size;
		mes .npc_name$;
		mes "Ofertas " + (.@start + 1) + "-" + ((.@start + .offer_page_size > .offer_count) ? .offer_count : (.@start + .offer_page_size)) + " de " + .offer_count + ":";
		for (.@i = 0; .@i < .offer_page_size; .@i++) {
			.@idx = .@start + .@i;
			if (.@idx >= .offer_count) break;
			mes (.@i + 1) + ") " + getitemname(.offer_id[.@idx]) + " - " + F_InsertComma(.offer_price[.@idx]) + " zeny c/u";
		}
		mes "Presupuesto restante: " + F_InsertComma(.daily_cap - #GARRETT_SPENT) + " zeny.";
		next;

		.@items = .offer_count - .@start;
		if (.@items > .offer_page_size) .@items = .offer_page_size;
		.@menu$ = "";
		for (.@i = 0; .@i < .@items; .@i++) {
			.@menu$ = .@menu$ + "Comprar " + (.@i + 1) + ":";
		}
		.@menu$ = .@menu$ + "Siguiente:Anterior:Volver";
		.@sel = select(.@menu$) - 1;

		if (.@sel < .@items) {
			.@item = .offer_id[.@start + .@sel];
			.@price = .offer_price[.@start + .@sel];
			callsub L_BuyOffer, .@item, .@price;
			return;
		}
		.@nav = .@sel - .@items;
		if (.@nav == 0) {
			if (.@page + 1 < .offer_pages) .@page++;
			continue;
		}
		if (.@nav == 1) {
			if (.@page > 0) .@page--;
			continue;
		}
		return;
	}

L_BuyOffer:
	.@item = getarg(0);
	.@price = getarg(1);
	.@today = gettime(DT_YYYYMMDD);

	mes .npc_name$;
	mes "¿Cuántos " + getitemname(.@item) + " me vendes?";
	mes "Tienes " + countitem(.@item) + " en el inventario.";
	next;

	input .@amount;
	if (.@amount <= 0) {
		mes .npc_name$;
		mes "Vamos, hablemos en serio.";
		next;
		return;
	}
	if (countitem(.@item) < .@amount) {
		mes .npc_name$;
		mes "No tienes tantos. Solo llevas " + countitem(.@item) + ".";
		next;
		return;
	}

	.@remaining = .daily_cap - #GARRETT_SPENT;
	.@max = .@remaining / .@price;
	if (.@max <= 0) {
		mes .npc_name$;
		mes "Con lo que me queda hoy no puedo comprar ni uno.";
		next;
		return;
	}
	if (.@amount > .@max) {
		mes .npc_name$;
		mes "Solo puedo comprar " + .@max + " con lo que me queda hoy.";
		next;
		.@amount = .@max;
	}

	.@total = .@price * .@amount;
	mes .npc_name$;
	mes .@amount + " x " + getitemname(.@item) + " = " + F_InsertComma(.@total) + " zeny. ¿Trato?";
	next;
	if (select("Trato hecho", "Paso") == 2) {
		mes .npc_name$;
		mes "La puerta sigue abierta.";
		next;
		return;
	}

	delitem .@item, .@amount;
	Zeny += .@total;
	#GARRETT_SPENT += .@total;
	#GARRETT_TOTAL_SPENT += .@total;
	#GARRETT_TRADES++;

	if (#GARRETT_KARMA_DAY != .@today) {
		OZRO_KARMA += 1;
		#GARRETT_KARMA_DAY = .@today;
	}

	mes .npc_name$;
	mes "Un placer hacer negocios.";
	if (#GARRETT_TRADES % .trades_for_coin == 0) {
		getitem "Ozro_Bronze_Coin", 1;
		mes "Por cierto... eres un sujeto agradable.";
		mes "Creo que esto te sirve más a ti que a mí.";
	}
	next;
	return;

L_MissionMenu:
	.@today = getarg(0);
	if (#GARRETT_MISSION_DAY == .@today) {
		mes .npc_name$;
		mes "Hoy ya no tengo misiones para ti. Vuelve mañana.";
		next;
		return;
	}

	.@page = 0;
	while (1) {
		.@start = .@page * .mission_page_size;
		mes .npc_name$;
		mes "Misiones " + (.@start + 1) + "-" + ((.@start + .mission_page_size > .mission_count) ? .mission_count : (.@start + .mission_page_size)) + " de " + .mission_count + ":";
		for (.@i = 0; .@i < .mission_page_size; .@i++) {
			.@idx = .@start + .@i;
			if (.@idx >= .mission_count) break;
			.@t$ = .tier_name$[.mission_tier[.@idx]];
			mes (.@i + 1) + ") " + .@t$ + ": " + getitemname(.mission_item[.@idx]) + " x" + .mission_qty[.@idx] + " - " + F_InsertComma(.mission_pay[.@idx]) + " zeny";
		}
		mes "Tienes 6 horas. Entregas parciales permitidas; si fallas, pierdes lo entregado.";
		next;

		.@items = .mission_count - .@start;
		if (.@items > .mission_page_size) .@items = .mission_page_size;
		.@menu$ = "";
		for (.@i = 0; .@i < .@items; .@i++) {
			.@menu$ = .@menu$ + "Misión " + (.@i + 1) + ":";
		}
		.@menu$ = .@menu$ + "Siguiente:Anterior:Volver";
		.@sel = select(.@menu$) - 1;

		if (.@sel < .@items) {
			.@idx = .@start + .@sel;
			callsub L_AcceptMission, .@idx, .@today;
			return;
		}
		.@nav = .@sel - .@items;
		if (.@nav == 0) {
			if (.@page + 1 < .mission_pages) .@page++;
			continue;
		}
		if (.@nav == 1) {
			if (.@page > 0) .@page--;
			continue;
		}
		return;
	}

L_AcceptMission:
	.@idx = getarg(0);
	.@today = getarg(1);
	.@item = .mission_item[.@idx];
	.@qty = .mission_qty[.@idx];
	.@pay = .mission_pay[.@idx];

	mes .npc_name$;
	mes "Misión " + .tier_name$[.mission_tier[.@idx]] + ":";
	mes "- " + getitemname(.@item) + " x" + .@qty;
	mes "- Pago: " + F_InsertComma(.@pay) + " zeny";
	mes "- Tiempo: 6 horas";
	mes "Entregas parciales permitidas. Si no completas, pierdes lo entregado.";
	next;

	if (select("Aceptar", "Rechazar") == 2) {
		mes .npc_name$;
		mes "Sin prisa. Los buenos tratos requieren calma.";
		next;
		return;
	}

	#GARRETT_MISSION_ITEM = .@item;
	#GARRETT_MISSION_QTY = .@qty;
	#GARRETT_MISSION_PAY = .@pay;
	#GARRETT_MISSION_ACTIVE = 1;
	#GARRETT_MISSION_DAY = .@today;
	#GARRETT_MISSION_END = gettimetick(2) + .mission_seconds;
	#GARRETT_MISSION_DELIVERED = 0;

	mes .npc_name$;
	mes "Aceptado. Tienes 6 horas.";
	mes "Puedes entregar por partes. Si no completas, pierdes lo entregado.";
	next;
	return;

L_MissionStatus:
	.@today = getarg(0);
	.@left = #GARRETT_MISSION_END - gettimetick(2);
	if (.@left <= 0) {
		#GARRETT_MISSION_ACTIVE = 0;
		#GARRETT_MISSION_DELIVERED = 0;
		#GARRETT_MISSION_DAY = .@today;
		mes .npc_name$;
		mes "Llegaste tarde.";
		mes "Los items entregados se perdieron. Así son los tratos conmigo.";
		next;
		return;
	}

	.@remaining = #GARRETT_MISSION_QTY - #GARRETT_MISSION_DELIVERED;
	.@hours = .@left / 3600;
	.@mins = (.@left % 3600) / 60;
	mes .npc_name$;
	mes "Misión activa:";
	mes "- " + getitemname(#GARRETT_MISSION_ITEM) + " x" + #GARRETT_MISSION_QTY;
	mes "Entregado: " + #GARRETT_MISSION_DELIVERED + "/" + #GARRETT_MISSION_QTY;
	mes "Restante: " + .@remaining;
	mes "Tiempo: " + .@hours + "h " + .@mins + "m";
	next;

	switch (select("Entregar items:Cancelar misión:Volver")) {
	case 1:
		if (countitem(#GARRETT_MISSION_ITEM) <= 0) {
			mes .npc_name$;
			mes "No tienes nada de lo que me pediste.";
			next;
			return;
		}
		mes .npc_name$;
		mes "¿Cuántos entregas? Tienes " + countitem(#GARRETT_MISSION_ITEM) + ", faltan " + .@remaining + ".";
		next;
		input .@amount;
		if (.@amount <= 0) {
			mes .npc_name$;
			mes "Vamos, hablemos en serio.";
			next;
			return;
		}
		if (.@amount > .@remaining) .@amount = .@remaining;
		if (countitem(#GARRETT_MISSION_ITEM) < .@amount) {
			mes .npc_name$;
			mes "No tienes tantos. Solo llevas " + countitem(#GARRETT_MISSION_ITEM) + ".";
			next;
			return;
		}
		delitem #GARRETT_MISSION_ITEM, .@amount;
		#GARRETT_MISSION_DELIVERED += .@amount;
		if (#GARRETT_MISSION_DELIVERED >= #GARRETT_MISSION_QTY) {
			Zeny += #GARRETT_MISSION_PAY;
			#GARRETT_MISSION_ACTIVE = 0;
			#GARRETT_MISSION_DELIVERED = 0;
			#GARRETT_MISSION_DAY = .@today;
			#GARRETT_MISSIONS_DONE++;
			#GARRETT_MISSION_EARNED += #GARRETT_MISSION_PAY;
			mes .npc_name$;
			mes "Trato hecho. Aquí tienes " + F_InsertComma(#GARRETT_MISSION_PAY) + " zeny.";
			mes "No tendré otra misión para ti hasta mañana.";
			next;
		} else {
			mes .npc_name$;
			mes "Entregaste " + .@amount + ". Faltan " + (#GARRETT_MISSION_QTY - #GARRETT_MISSION_DELIVERED) + ".";
			next;
		}
		return;
	case 2:
		mes .npc_name$;
		mes "Si cancelas, pierdes lo entregado y no habrá otra misión hasta mañana.";
		next;
		if (select("Sí, cancelar", "No, continuar") == 1) {
			#GARRETT_MISSION_ACTIVE = 0;
			#GARRETT_MISSION_DELIVERED = 0;
			#GARRETT_MISSION_DAY = .@today;
			mes .npc_name$;
			mes "Está bien. La basura puede esperar.";
			next;
		}
		return;
	case 3:
		return;
	}

L_ResetDay:
	.@today = getarg(0);
	if (#GARRETT_DAY == .@today) return;
	#GARRETT_DAY = .@today;
	#GARRETT_SPENT = 0;
	return;
