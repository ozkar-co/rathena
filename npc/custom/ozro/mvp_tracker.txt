//===== OzRo Script ==========================================
//= MVP Tracker
//===== Description: =========================================
//= Registra golpes finales, asistencias de party y puntos de Alma.
//= Variables existentes que NO se deben renombrar:
//=   MVP_KILL_<id>  -> golpes finales
//=   MVP_SUPP_<id>  -> asistencias de party
//= Variables nuevas:
//=   MVP_SOUL_POINTS -> puntos pendientes de reclamar
//=============================================================

-	script	MVP_Tracker	-1,{
OnNPCKillEvent:
	if (killedrid <= 0) end;

	.@is_mvp = 0;
	if (getmonsterinfo(killedrid, MOB_MVPEXP) > 0) .@is_mvp = 1;
	if (getmonsterinfo(killedrid, MOB_MODE) & MD_MVP) .@is_mvp = 1;
	if (!.@is_mvp) end;

	.@mvp_id = killedrid;
	.@killer_aid = getcharid(3);
	.@party_id = getcharid(1);
	.@map$ = strcharinfo(3);

	// Solo los MVPs naturales/guild entregan puntos de Alma.
	.@natural = 0;
	for (.@i = 0; .@i < .natural_count; .@i++) {
		if (.natural_mvp_id[.@i] == .@mvp_id) {
			.@natural = 1;
			break;
		}
	}

	.@points = 0;
	if (.@natural) {
		.@mexp = getmonsterinfo(.@mvp_id, MOB_MVPEXP);
		.@level = getmonsterinfo(.@mvp_id, MOB_LV);
		if (.@mexp > 0) {
			if (.@mexp < 500000) .@points = 1;
			else if (.@mexp < 1500000) .@points = 5;
			else .@points = 10;
		} else {
			if (.@level < 120) .@points = 1;
			else if (.@level < 180) .@points = 5;
			else .@points = 10;
		}
	}

	// Golpe final para el jugador que mato al MVP.
	.@kill_var$ = "MVP_KILL_" + .@mvp_id;
	setd .@kill_var$, getd(.@kill_var$) + 1;
	if (.@points > 0) MVP_SOUL_POINTS = MVP_SOUL_POINTS + .@points;

	// Asistencias para los miembros de la party en el mismo mapa.
	if (.@party_id > 0) {
		getpartymember .@party_id, 2;
		for (.@i = 0; .@i < $@partymembercount; .@i++) {
			if ($@partymemberaid[.@i] == .@killer_aid) continue;
			if (attachrid($@partymemberaid[.@i])) {
				if (strcharinfo(3) == .@map$) {
					.@supp_var$ = "MVP_SUPP_" + .@mvp_id;
					setd .@supp_var$, getd(.@supp_var$) + 1;
					if (.@points > 0) MVP_SOUL_POINTS = MVP_SOUL_POINTS + .@points;
				}
				detachrid;
			}
		}
	}
	end;

OnInit:
	setarray .natural_mvp_id[0], 1511, 2362, 2249, 1785, 1039, 3505, 2068, 20648, 21301, 20422, 20421, 2255, 1272, 20943, 1719, 1046, 1389, 1112, 1115, 1658, 1418, 1871, 2253, 2251, 1768, 2165, 1086, 1885, 1252, 1832, 20601, 1734, 2442, 2202, 1688, 2156, 1147, 1059, 1150, 2483, 1087, 1190, 1038, 1157, 1159, 20934, 20381, 20419, 1623, 1492, 2087, 3741, 1251, 1583, 2441, 20928, 3074, 1312, 1751, 3633, 1685, 1630, 1917;
	.natural_count = getarraysize(.natural_mvp_id);
	return;
}
