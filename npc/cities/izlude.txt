-	script	::Sailor_izlude	-1,{
	mes "[Marinero]";
	mes "¡Atención, atención!";
	mes "¡Último llamado para abordar!";
	mes "¡Barcos con destino a";
	mes "Byalan y Alberta!";
	mes "¡No se queden atrás!";
	next;
	switch(select("Isla Byalan -> 150 Zeny.:Puerto de Alberta -> 500 Zeny.:Cancelar.")) {
	case 1:
		if (Zeny < 150) {
			mes "[Marinero]";
			mes "Lo siento.";
			mes "El pasaje cuesta";
			mes "150 Zeny.";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[Marinero]";
			mes "Lo siento.";
			mes "El pasaje cuesta";
			mes "500 Zeny.";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
